### Step 81: Implement Compliance Auditing

**Objective**: Ensure DEA-C01 compliance requirements are met.

**Actions**:
Create `python/src/utils/compliance_checker.py`:
```python
#!/usr/bin/env python3
"""
DEA-C01 Compliance Checker
"""
import boto3
import json
from typing import List, Dict, Tuple

class ComplianceChecker:
    def __init__(self, region: str):
        self.region = region
        self.emr = boto3.client('emr', region_name=region)
        self.kms = boto3.client('kms', region_name=region)
        self.s3 = boto3.client('s3', region_name=region)
    
    def check_encryption_at_rest(self, cluster_id: str) -> Tuple[bool, str]:
        """Verify encryption at rest compliance"""
        cluster = self.emr.describe_cluster(ClusterId=cluster_id)['Cluster']
        
        if 'SecurityConfiguration' not in cluster:
            return False, "No security configuration"
        
        config_name = cluster['SecurityConfiguration']
        config = self.emr.describe_security_configuration(Name=config_name)
        config_json = json.loads(config['SecurityConfiguration'])
        
        at_rest = config_json.get('EncryptionConfiguration', {}).get('EnableAtRestEncryption', False)
        
        if not at_rest:
            return False, "At-rest encryption not enabled"
        
        return True, "At-rest encryption compliant"
    
    def check_encryption_in_transit(self, cluster_id: str) -> Tuple[bool, str]:
        """Verify encryption in transit compliance"""
        cluster = self.emr.describe_cluster(ClusterId=cluster_id)['Cluster']
        config_name = cluster['SecurityConfiguration']
        config = self.emr.describe_security_configuration(Name=config_name)
        config_json = json.loads(config['SecurityConfiguration'])
        
        in_transit = config_json.get('EncryptionConfiguration', {}).get('EnableInTransitEncryption', False)
        
        if not in_transit:
            return False, "In-transit encryption not enabled"
        
        return True, "In-transit encryption compliant"
    
    def check_kms_key_rotation(self, key_id: str) -> Tuple[bool, str]:
        """Verify KMS key rotation is enabled"""
        try:
            rotation = self.kms.get_key_rotation_status(KeyId=key_id)
            if rotation['KeyRotationEnabled']:
                return True, "KMS key rotation enabled"
            return False, "KMS key rotation disabled"
        except Exception as e:
            return False, f"Error checking rotation: {e}"
    
    def check_s3_bucket_policies(self, bucket: str) -> Tuple[bool, str]:
        """Verify S3 bucket has proper policies"""
        try:
            # Check public access block
            public_access = self.s3.get_public_access_block(Bucket=bucket)
            config = public_access['PublicAccessBlockConfiguration']
            
            if not all([
                config['BlockPublicAcls'],
                config['BlockPublicPolicy'],
                config['IgnorePublicAcls'],
                config['RestrictPublicBuckets']
            ]):
                return False, "Public access not fully blocked"
            
            # Check encryption
            try:
                encryption = self.s3.get_bucket_encryption(Bucket=bucket)
                return True, "Bucket properly secured"
            except:
                return False, "Bucket encryption not configured"
                
        except Exception as e:
            return False, f"Error checking bucket: {e}"
    
    def check_iam_least_privilege(self, role_name: str) -> Tuple[bool, str]:
        """Verify IAM roles follow least privilege"""
        iam = boto3.client('iam', region_name=self.region)
        
        # Get role policies
        policies = iam.list_attached_role_policies(RoleName=role_name)
        
        # Check for overly permissive policies
        for policy in policies['AttachedPolicies']:
            if policy['PolicyArn'].endswith('AdministratorAccess'):
                return False, "Admin access policy attached"
        
        return True, "IAM role follows least privilege"
    
    def check_cloudtrail_logging(self) -> Tuple[bool, str]:
        """Verify CloudTrail is enabled"""
        cloudtrail = boto3.client('cloudtrail', region_name=self.region)
        
        trails = cloudtrail.describe_trails()['trailList']
        
        if not trails:
            return False, "No CloudTrail trails configured"
        
        active_trails = [t for t in trails if t.get('IsMultiRegionTrail', False)]
        
        if not active_trails:
            return False, "No multi-region trail configured"
        
        return True, "CloudTrail properly configured"
    
    def run_full_audit(self, cluster_id: str, key_id: str, 
                       buckets: List[str], roles: List[str]) -> Dict:
        """Run complete compliance audit"""
        results = {
            'timestamp': datetime.utcnow().isoformat(),
            'cluster_id': cluster_id,
            'checks': {}
        }
        
        # Run all checks
        checks = [
            ('encryption_at_rest', self.check_encryption_at_rest(cluster_id)),
            ('encryption_in_transit', self.check_encryption_in_transit(cluster_id)),
            ('kms_rotation', self.check_kms_key_rotation(key_id)),
            ('cloudtrail', self.check_cloudtrail_logging()),
        ]
        
        for bucket in buckets:
            checks.append((f's3_bucket_{bucket}', self.check_s3_bucket_policies(bucket)))
        
        for role in roles:
            checks.append((f'iam_role_{role}', self.check_iam_least_privilege(role)))
        
        # Process results
        for name, (passed, message) in checks:
            results['checks'][name] = {
                'passed': passed,
                'message': message
            }
        
        # Calculate compliance score
        total = len(checks)
        passed = sum(1 for _, (p, _) in checks if p)
        results['compliance_score'] = (passed / total) * 100
        results['overall_status'] = 'COMPLIANT' if passed == total else 'NON_COMPLIANT'
        
        return results

if __name__ == "__main__":
    import argparse
    
    parser = argparse.ArgumentParser()
    parser.add_argument('--cluster-id', required=True)
    parser.add_argument('--key-id', required=True)
    parser.add_argument('--buckets', nargs='+', required=True)
    parser.add_argument('--roles', nargs='+', required=True)
    parser.add_argument('--region', default='us-east-1')
    parser.add_argument('--output', default='compliance-report.json')
    args = parser.parse_args()
    
    checker = ComplianceChecker(args.region)
    results = checker.run_full_audit(
        args.cluster_id,
        args.key_id,
        args.buckets,
        args.roles
    )
    
    # Print summary
    print(f"\n{'='*60}")
    print(f"DEA-C01 COMPLIANCE AUDIT REPORT")
    print(f"{'='*60}")
    print(f"Cluster: {args.cluster_id}")
    print(f"Status: {results['overall_status']}")
    print(f"Compliance Score: {results['compliance_score']:.1f}%")
    print(f"{'='*60}\n")
    
    for check, result in results['checks'].items():
        status = '✓' if result['passed'] else '✗'
        print(f"{status} {check}: {result['message']}")
    
    # Save detailed report
    with open(args.output, 'w') as f:
        json.dump(results, f, indent=2)
    
    print(f"\nDetailed report saved to: {args.output}")
    
    # Exit code based on compliance
    exit(0 if results['overall_status'] == 'COMPLIANT' else 1)
```

#!/usr/bin/env python3
"""
Data Quality Framework for EMR
"""
from pyspark.sql import SparkSession
from typing import Dict, List, Any

class DataQualityChecker:
    def __init__(self, spark: SparkSession):
        self.spark = spark
        self.results = []
    
    def check_null_values(self, df, columns: List[str], threshold: float = 0.05):
        """Check for excessive null values"""
        total_rows = df.count()
        
        for col in columns:
            null_count = df.filter(df[col].isNull()).count()
            null_percentage = null_count / total_rows
            
            passed = null_percentage <= threshold
            self.results.append({
                'check': f'null_check_{col}',
                'passed': passed,
                'metric': null_percentage,
                'threshold': threshold
            })
        
        return self
    
    def check_uniqueness(self, df, columns: List[str]):
        """Check for duplicate records"""
        total = df.count()
        distinct = df.select(columns).distinct().count()
        
        passed = total == distinct
        self.results.append({
            'check': 'uniqueness_check',
            'passed': passed,
            'duplicates': total - distinct
        })
        
        return self
    
    def check_schema(self, df, expected_schema: Dict[str, str]):
        """Verify DataFrame schema matches expected"""
        actual_schema = {field.name: str(field.dataType) for field in df.schema.fields}
        
        matches = all(
            actual_schema.get(col) == dtype
            for col, dtype in expected_schema.items()
        )
        
        self.results.append({
            'check': 'schema_check',
            'passed': matches,
            'expected': expected_schema,
            'actual': actual_schema
        })
        
        return self
    
    def check_data_freshness(self, df, timestamp_col: str, max_age_hours: int = 24):
        """Check if data is fresh"""
        from pyspark.sql.functions import max as spark_max, current_timestamp, col
        from datetime import timedelta
        
        latest = df.select(spark_max(timestamp_col)).collect()[0][0]
        age = (datetime.now() - latest).total_seconds() / 3600
        
        passed = age <= max_age_hours
        self.results.append({
            'check': 'freshness_check',
            'passed': passed,
            'age_hours': age,
            'threshold_hours': max_age_hours
        })
        
        return self
    
    def generate_report(self) -> Dict[str, Any]:
        """Generate quality report"""
        total_checks = len(self.results)
        passed_checks = sum(1 for r in self.results if r['passed'])
        
        return {
            'total_checks': total_checks,
            'passed_checks': passed_checks,
            'failed_checks': total_checks - passed_checks,
            'quality_score': (passed_checks / total_checks) * 100 if total_checks > 0 else 0,
            'details': self.results
        }

# Example usage
if __name__ == "__main__":
    spark = SparkSession.builder.appName("DataQuality").getOrCreate()
    
    # Load data
    df = spark.read.parquet("s3://bucket/path/")
    
    # Run checks
    checker = DataQualityChecker(spark)
    report = (checker
              .check_null_values(df, ['id', 'name', 'value'])
              .check_uniqueness(df, ['id'])
              .check_schema(df, {'id': 'IntegerType', 'name': 'StringType'})
              .check_data_freshness(df, 'timestamp')
              .generate_report())
    
    print(f"Quality Score: {report['quality_score']:.1f}%")
    
    # Fail if quality below threshold
